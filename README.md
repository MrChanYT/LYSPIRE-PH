  <script>
    // ─── Modal ──────────────────────────────────────────────────────────
    const modal        = document.getElementById('applyModal');
    const modalCard    = modal.querySelector('.modal-card');
    const openBtns     = document.querySelectorAll('[data-open-modal]');
    const closeBtns    = document.querySelectorAll('[data-close-modal]');
    // FIX: include #fab so it is also made inert when modal opens
    const bgLandmarks  = document.querySelectorAll('body > nav, body > main, body > footer, #fab');
    let lastFocused    = null;

    function focusableEls(){ return modalCard.querySelectorAll('a[href], button:not([disabled])'); }

    function openModal(e){
      lastFocused = e.currentTarget;
      modal.classList.add('show');
      modal.setAttribute('aria-hidden','false');
      document.body.classList.add('modal-open');
      bgLandmarks.forEach(el => el.setAttribute('inert',''));
      modalCard.focus();
      document.addEventListener('keydown', trapFocus);
    }

    function closeModal(){
      modal.classList.remove('show');
      modal.setAttribute('aria-hidden','true');
      document.body.classList.remove('modal-open');
      bgLandmarks.forEach(el => el.removeAttribute('inert'));
      document.removeEventListener('keydown', trapFocus);
      if(lastFocused) lastFocused.focus();
    }

    function trapFocus(e){
      if(e.key === 'Escape'){ closeModal(); return; }
      if(e.key !== 'Tab') return;
      const els   = Array.from(focusableEls());
      if(!els.length) return;
      const first = els[0];
      const last  = els[els.length - 1];
      if(e.shiftKey && document.activeElement === first){ e.preventDefault(); last.focus(); }
      else if(!e.shiftKey && document.activeElement === last){ e.preventDefault(); first.focus(); }
    }

    openBtns.forEach(btn => btn.addEventListener('click', openModal));

    // Close buttons — <a> elements navigate, so defer the modal close slightly
    closeBtns.forEach(btn => {
      const isLink = btn.tagName === 'A';
      btn.addEventListener('click', isLink ? () => setTimeout(closeModal, 60) : closeModal);
    });

    modal.addEventListener('click', e => { if(e.target === modal) closeModal(); });

    // ─── Toast ──────────────────────────────────────────────────────────
    const toastEl = document.getElementById('toast');
    let toastTimer;
    function showToast(msg, ms = 2400){
      toastEl.textContent = msg;
      toastEl.setAttribute('aria-hidden','false');
      toastEl.classList.add('show');
      clearTimeout(toastTimer);
      toastTimer = setTimeout(() => {
        toastEl.classList.remove('show');
        toastEl.setAttribute('aria-hidden','true');
      }, ms);
    }

    // ─── Copy email to clipboard ────────────────────────────────────────
    document.getElementById('copyEmailBtn').addEventListener('click', async () => {
      try {
        await navigator.clipboard.writeText('hr.marche01@gmail.com');
        showToast('✓ Email address copied!');
      } catch {
        showToast('hr.marche01@gmail.com — copy it manually.');
      }
    });

    // ─── Share job ──────────────────────────────────────────────────────
    document.getElementById('shareBtn').addEventListener('click', async () => {
      const data = {
        title: 'LYSPIRE PH | Hiring Repacker / Warehouse Staff',
        text:  'LYSPIRE PH is hiring Repacker / Warehouse Staff in Carmona, Cavite. Limited slots!',
        url:   location.href
      };
      if(navigator.share && navigator.canShare && navigator.canShare(data)){
        try { await navigator.share(data); } catch {}
      } else {
        try {
          await navigator.clipboard.writeText(location.href);
          showToast('✓ Link copied — share it anywhere!');
        } catch {
          showToast('Copy the page URL to share this job.');
        }
      }
    });

    // ─── FAQ accordion ──────────────────────────────────────────────────
    document.querySelectorAll('.faq-q').forEach(btn => {
      btn.addEventListener('click', () => {
        const item   = btn.closest('.faq-item');
        const isOpen = item.classList.contains('open');
        // collapse all
        document.querySelectorAll('.faq-item.open').forEach(i => {
          i.classList.remove('open');
          i.querySelector('.faq-q').setAttribute('aria-expanded','false');
        });
        // expand clicked (if it was closed)
        if(!isOpen){
          item.classList.add('open');
          btn.setAttribute('aria-expanded','true');
        }
      });
    });

    // ─── Reading progress bar ────────────────────────────────────────────
    const progressBar = document.getElementById('progressBar');
    function updateProgress(){
      const scrolled  = window.scrollY;
      const total     = document.documentElement.scrollHeight - window.innerHeight;
      progressBar.style.width = total > 0 ? (scrolled / total * 100) + '%' : '0%';
    }

    // ─── Active nav link highlighting ────────────────────────────────────
    const sections  = Array.from(document.querySelectorAll('section[id]'));
    const navLinks  = document.querySelectorAll('.nav-links a[data-nav]');
    function updateNav(){
      let active = '';
      sections.forEach(s => { if(window.scrollY + 130 >= s.offsetTop) active = s.id; });
      navLinks.forEach(a => a.classList.toggle('active', a.dataset.nav === active));
    }

    // ─── Floating Apply button (mobile) ─────────────────────────────────
    const fab = document.getElementById('fab');
    function updateFab(){
      const show = window.scrollY > 360;
      fab.classList.toggle('visible', show);
      fab.setAttribute('aria-hidden', show ? 'false' : 'true');
      fab.tabIndex = show ? 0 : -1;
    }

    // ─── Unified scroll handler ──────────────────────────────────────────
    function onScroll(){
      updateProgress();
      updateNav();
      updateFab();
    }
    window.addEventListener('scroll', onScroll, { passive: true });
    onScroll(); // run once on load
  </script>
